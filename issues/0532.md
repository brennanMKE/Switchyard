## Description

The visible half of guide §11 decision 41:

- **`RepositorySidebarView`** — the Remotes section groups by remote
  (`RepositorySidebarView.remoteGroups(remotes:branches:query:)`, pure, tested): each configured
  remote's row (`server.rack` label with the name, its fetch URL on a caption line under it,
  `remoteHelpText` as help — the push URLs only when they differ) followed by its
  `refs/remotes/<name>/*` rows, which keep their `origin/x` labels, #0401's click and #0510's menu;
  then the remote-tracking branches no configured remote owns, as before. While filtering, a remote
  shows when its name or any of its branches match. The section now shows while unfiltered even
  with no remotes — a "No remotes" row — so **Add Remote…** (on the "Remotes" header's menu, that
  row's, and every remote row's) is always reachable. A remote row's menu: **Fetch “x”**, **Prune
  “x”**, Edit URL…, Rename Remote…, Remove Remote…, Add Remote… — all disabled while anything runs.
  New init parameter `onRemoteCommand: ((RemoteMenuCommand) -> Void)? = nil` (defaulted, so previews
  and `RerereSurfaceTests` compile unchanged; `nil` hides the items).
- **`ContentView`** — `runningRemoteAction` (folded into `isBusy`, the Edit menu's busy check,
  `runJournal`'s guard and the progress line), `remoteSheet`, `pendingRemoteRemoval`, the
  `RemoteActionDialogs` modifier, `handleRemoteCommand(_:)` and `runRemoteAction(_:)`.
  - `handleRemoteCommand`: Add and Rename open the sheet at once; **Edit URL…** first reads the
    configured URL (`loadConfiguredURL`) and **Remove Remote…** first reads what goes
    (`loadRemoteRemovalImpact`) — neither read sets a busy flag, so the question it then presents
    is never dropped by `runRemoteAction`'s `!isBusy` guard (#0512's lesson: never present a
    follow-up while a busy flag is set). A failed read is the failure alert.
  - `runRemoteAction`: runs the action and then its `followUp` (Add's Fetch) as separate steps;
    a network step (Fetch, Prune) is the toolbar's `remoteTask` while it runs, so the progress
    line's existing **Cancel** reaches it (#0458). A failure stops the chain. Then
    `refreshAfterMutation` (which re-reads the sidebar, the toolbar's remote names and the journal
    listing), clear `runningRemoteAction`, and only then present the failure alert.

Every state write is in a button or menu action, or after an `await` in a task (event-origin —
swift-guidance's Observation rules); nothing is written from `body`.

Anchors below are unique in `main` at f72cc2aa. `RepositorySidebarView.swift`'s `remotesPrefix` is
the existing `private nonisolated static let`, readable from the new `nonisolated` static.

    public nonisolated static func remoteGroups(
        remotes: [RemoteConfig.Remote], branches: [RefSnapshot.Entry], query: String
    ) -> [RemoteGroup] {
        let filtering = !query.trimmingCharacters(in: .whitespaces).isEmpty
        var owned = Set<String>()

    /// #0532: a remote row's help text: where it fetches from and pushes to.
    public nonisolated static func remoteHelpText(_ remote: RemoteConfig.Remote) -> String {
        let fetch = "Fetch: " + (remote.fetchURL ?? "no URL")
        guard remote.pushDiffers else { return fetch }
        let push = remote.pushURLs.isEmpty ? ["no URL"] : remote.pushURLs
        return fetch + "\n" + push.map { "Push: " + $0 }.joined(separator: "\n")
    }

    // MARK: - #0532: remote management

    /// #0532: a Remotes-section menu item (guide §11 decision 41). Add and
    /// Rename open the sheet at once; Edit URL… reads the configured URL
    /// and Remove Remote… reads what it would delete before presenting —
    /// neither read sets a busy flag, so the question's button is never
    /// dropped by `runRemoteAction`'s guard (#0512's lesson).
    private func handleRemoteCommand(_ command: RemoteMenuCommand) {
        guard let repositoryPath, !isBusy, !journalRunning else { return }
        switch command {
        case .add:
            remoteSheet = .add
        case let .rename(remote):
            remoteSheet = .rename(remote: remote)
        case let .run(action):
            runRemoteAction(action)
        case let .editURL(remote, pushURLs):
            Task {
                do {
                    let url = try await loadConfiguredURL(remote: remote, at: repositoryPath)
                    remoteSheet = .editURL(remote: remote, url: url, pushURLs: pushURLs)
                } catch {
                    actionFailure = RemoteAction.setURL(remote: remote, url: "").failure(for: error)
                }
            }
        case let .remove(remote):
            Task {
                do {
                    let impact = try await loadRemoteRemovalImpact(remote: remote, at: repositoryPath)
                    pendingRemoteRemoval = RemoteRemovalConfirmation(remote: remote, impact: impact)
                } catch {
                    actionFailure = RemoteAction.remove(remote: remote).failure(for: error)
                }
            }
        }
    }

    /// #0532: runs one remote action and its follow-up (Add's Fetch), then
    /// refreshes in place. Fetch and Prune are the toolbar's `remoteTask`
    /// while they run, so the progress line's Cancel reaches them (#0458).
    /// A failure stops the chain; its alert is presented after the refresh
    /// and after `runningRemoteAction` clears.
    private func runRemoteAction(_ action: RemoteAction) {
        guard let repositoryPath, !isBusy, !journalRunning else { return }
        runningRemoteAction = action
        Task {
            var failure: CommitActionFailure?
            var step: RemoteAction? = action
            while let current = step {
                runningRemoteAction = current
                let work = Task { try await performRemoteAction(current, at: repositoryPath) }
                if current.usesNetwork { remoteTask = work }
                do {
                    try await work.value
                    step = current.followUp
                } catch {
                    failure = current.failure(for: error)
                    step = nil
                }
                remoteTask = nil
            }
            await refreshAfterMutation { _, _ in nil }
            runningRemoteAction = nil
            if let failure { actionFailure = failure }
        }
    }
