## Description

The VM spike for guide §11 decision 44, on a new fixture
(`scripts/uitest-fixtures/make-large-history-fixture.sh`): 6,002 commits and 1,000 lightweight
tags, written with one `git fast-import` stream. **Measured by running the script on the host into
the scratchpad** (git 2.54.0; the app was not launched): 2.9 s wall; `git rev-list --count --all` →
`6002`; `git tag | wc -l` → `1000`; `git log --oneline -1` → `large commit 6000`; `git log
--topo-order -5000 --format=%s HEAD --branches --remotes` (History's walk) holds 20 subjects
containing `needle` and ends at `large commit 1003`; `git log -1 --format=%s rel-5994` → `large
commit 5994`; the tree is clean.

The spike, in one test: open → History shows `large commit 6000` → type `needle` in the filter →
**"20 matches"** → replace with `rel-5994` → **"1 match"** (only the tag chip matches: the chips come
from `HistoryIndex`, so this fails if the index is missing or built without refs) → ⌘G selects
`large commit 5994`, screenshot `large-history-tag-match` → clear the filter, select the tip, **↓**
→ `large commit 5990` (the nine commits between fold into one row, #0427, so the next node in the
lane is the fork point of `large-topic`) → **↑** → `large commit 6000`, screenshot
`large-history-arrows`.

The seconds from launch to the tip row and from each last keystroke to its count are attached as
text (`large-history-timings`) and printed as `SPIKE-0555 TIMINGS` — **for the record, never
asserted** (CLAUDE.md). Every wait is a bounded `waitForExistence`.

It launches through the existing `launchWithRemoteFixture(_:)` (#0457) and uses `matchCount(_:)`
(#0525's extension in `0525HistorySearchUITests.swift`, same test target), `sidebarFilterField()`,
`historyRows(containing:)` and `selectHistoryRow(subject:)`, so `UITestSupport.swift` is not
edited. The fixture's author is `Switchyard UI Test`, which `historyRows(containing:)` requires.

## The files

### 1. `scripts/uitest-fixtures/make-large-history-fixture.sh` — new file, exactly this

```sh
#!/bin/zsh
# make-large-history-fixture.sh <path> — the large-history UI fixture
# (#0555, guide §11 decision 44).
#
# Run inside the Tart guest by scripts/run-ui-tests-vm.sh; never shared,
# never host-live. Written with one `git fast-import` stream so 6,002
# commits and 1,000 tags take about a second.
#
#   large-main   HEAD: "large commit 0001".."6000", a minute apart, the
#                newest now -- every tip is inside #0429's two-week filter.
#                Commits 0125, 0375, ... (every 250th from 125) end " needle":
#                24 of them, 20 inside History's newest 5,000 (1125..5875).
#   large-topic  "topic commit 1", "topic commit 2" off "large commit 5990"
#   rel-NNNN     a lightweight tag on every 6th commit (rel-0006..rel-6000)
set -euo pipefail
repo="$1"
git init -q --initial-branch=large-main "$repo"
cd "$repo"
git config user.email uitest@example.invalid
git config user.name 'Switchyard UI Test'
git config commit.gpgsign false
now=$(date +%s)
{
  for n in {1..6000}; do
    num=$(printf '%04d' $n)
    subject="large commit $num"
    (( n % 250 == 125 )) && subject="$subject needle"
    when=$(( now - (6000 - n) * 60 ))
    print "commit refs/heads/large-main"
    print "mark :$n"
    print "author Switchyard UI Test <uitest@example.invalid> $when +0000"
    print "committer Switchyard UI Test <uitest@example.invalid> $when +0000"
    print "data ${#subject}"
    print -r -- "$subject"
    (( n > 1 )) && print "from :$(( n - 1 ))"
    print "M 644 inline large.txt"
    print "data ${#num}"
    print -r -- "$num"
    print ""
  done
  for t in 1 2; do
    subject="topic commit $t"
    when=$(( now - 30 + t ))
    print "commit refs/heads/large-topic"
    print "mark :$(( 6000 + t ))"
    print "author Switchyard UI Test <uitest@example.invalid> $when +0000"
    print "committer Switchyard UI Test <uitest@example.invalid> $when +0000"
    print "data ${#subject}"
    print -r -- "$subject"
    (( t == 1 )) && print "from :5990" || print "from :6001"
    print "M 644 inline topic.txt"
    print "data 1"
    print "$t"
    print ""
  done
  for n in {6..6000..6}; do
    print "reset refs/tags/rel-$(printf '%04d' $n)"
    print "from :$n"
    print ""
  done
} | git fast-import --quiet
git checkout -q -f large-main
```

### 2. `SwitchyardUITests/0555LargeHistoryUITests.swift` — new file, exactly this

```swift
import XCTest

/// #0555: the large-history fixture
/// (scripts/uitest-fixtures/make-large-history-fixture.sh) — keep the two in sync.
enum UITestLargeHistoryFixture {
    static let repositoryPath = "/Users/admin/uitest-large-repo"
    /// HEAD's tip, and the fork point one arrow press below it (the nine
    /// commits between fold into one row).
    static let tipSubject = "large commit 6000"
    static let forkSubject = "large commit 5990"
    /// 20 loaded commits' subjects end with it.
    static let needle = "needle"
    /// A tag only on "large commit 5994": found through its ref chip.
    static let tag = "rel-5994"
    static let tagSubject = "large commit 5994"
}

/// #0555: History on 5,000 loaded commits and 1,000 tags (guide §11
/// decision 44). The filter counts message matches and finds a commit by a
/// tag chip — the chips and search text come from `HistoryIndex` — and the
/// arrow keys step through HEAD's lane. The seconds from the last keystroke
/// to the count appearing are attached as text, for the record: they are
/// never asserted (CLAUDE.md: never assert wall-clock time).
final class Spike0555LargeHistoryUITests: XCTestCase {
    @MainActor
    func testFilterAndArrowsOnALargeHistory() {
        let app = XCUIApplication()
        let launched = Date()
        app.launchWithRemoteFixture(UITestLargeHistoryFixture.repositoryPath)
        let window = app.windows.firstMatch
        let tip = app.historyRows(containing: UITestLargeHistoryFixture.tipSubject).firstMatch
        XCTAssertTrue(tip.waitForExistence(timeout: 60), "History does not show the large repository's tip")
        var timings = ["open to tip row: \(seconds(since: launched))"]

        let filter = app.sidebarFilterField()
        XCTAssertTrue(filter.waitForExistence(timeout: 30), "no filter field")
        filter.click()
        filter.typeText(UITestLargeHistoryFixture.needle)
        var typed = Date()
        XCTAssertTrue(app.matchCount("20 matches").waitForExistence(timeout: 60),
                      "the filter does not count the 20 loaded commits whose subject says needle")
        timings.append("\"needle\" to \"20 matches\": \(seconds(since: typed))")

        filter.typeKey("a", modifierFlags: .command)
        filter.typeText(UITestLargeHistoryFixture.tag)
        typed = Date()
        XCTAssertTrue(app.matchCount("1 match").waitForExistence(timeout: 60),
                      "the filter does not find the commit tagged rel-5994 by its chip")
        timings.append("\"rel-5994\" to \"1 match\": \(seconds(since: typed))")
        app.typeKey("g", modifierFlags: .command)
        let tagged = app.staticTexts.matching(NSPredicate(
            format: "value == %@ OR label == %@",
            UITestLargeHistoryFixture.tagSubject, UITestLargeHistoryFixture.tagSubject)).firstMatch
        XCTAssertTrue(tagged.waitForExistence(timeout: 30), "⌘G did not select the tagged commit")
        let filtered = XCTAttachment(screenshot: window.screenshot())
        filtered.name = "large-history-tag-match"
        filtered.lifetime = .keepAlways
        add(filtered)

        // Clear the filter, then step down HEAD's lane and back up.
        filter.click()
        filter.typeKey("a", modifierFlags: .command)
        filter.typeKey(.delete, modifierFlags: [])
        app.selectHistoryRow(subject: UITestLargeHistoryFixture.tipSubject)
        app.typeKey(.downArrow, modifierFlags: [])
        let fork = app.staticTexts.matching(NSPredicate(
            format: "value == %@ OR label == %@",
            UITestLargeHistoryFixture.forkSubject, UITestLargeHistoryFixture.forkSubject)).firstMatch
        XCTAssertTrue(fork.waitForExistence(timeout: 30), "↓ did not step from the tip to the fork point")
        app.typeKey(.upArrow, modifierFlags: [])
        let tipHeadline = app.staticTexts.matching(NSPredicate(
            format: "value == %@ OR label == %@",
            UITestLargeHistoryFixture.tipSubject, UITestLargeHistoryFixture.tipSubject)).firstMatch
        XCTAssertTrue(tipHeadline.waitForExistence(timeout: 30), "↑ did not step back to the tip")
        let stepped = XCTAttachment(screenshot: window.screenshot())
        stepped.name = "large-history-arrows"
        stepped.lifetime = .keepAlways
        add(stepped)

        let record = XCTAttachment(string: timings.joined(separator: "\n"))
        record.name = "large-history-timings"
        record.lifetime = .keepAlways
        add(record)
        print("SPIKE-0555 TIMINGS\n" + timings.joined(separator: "\n"))
    }

    private func seconds(since start: Date) -> String {
        String(format: "%.2f s", Date().timeIntervalSince(start))
    }
}
```
