### The spike app

app covers all six, one tab each. Every action appends to a numbered log, so a double-fire shows as
two lines.

Create it in the spike round's worktree under `build/spike-m7/` (AGENTS.md's scratch rule — never

```swift
// M7SpikesApp.swift -- one throwaway app exercising the six SwiftUI
// behaviours #0380's spikes verify. Every action appends to a visible,
// numbered log, so a double-fire shows as two lines.

import AppKit
import SwiftUI

@Observable
final class SpikeLog {
    private(set) var lines: [String] = []

    func record(_ line: String) {
        lines.append("\(lines.count + 1). \(line)")
    }
}

@main
struct M7SpikesApp: App {
    @State private var log = SpikeLog()

    init() {
        // An unbundled `swift run` executable starts as a background process;
        // without this it gets no menu bar and no key window.
        NSApplication.shared.setActivationPolicy(.regular)
    }

    var body: some Scene {
        WindowGroup("M7 Spikes") {
            SpikeRoot()
                .environment(log)
                .onAppear { NSApp.activate() }
        }
        .commands {
            CommandMenu("Spike") {
                Button("Shared ⌥⌘J") { log.record("menu bar: Shared ⌥⌘J") }
                    .keyboardShortcut("j", modifiers: [.command, .option])
                Button("Move Up") { log.record("menu bar: ⌥⌘↑") }
                    .keyboardShortcut(.upArrow, modifiers: [.command, .option])
                Button("Move Down") { log.record("menu bar: ⌥⌘↓") }
                    .keyboardShortcut(.downArrow, modifiers: [.command, .option])
                Divider()
                Button("Disabled With Help") {}
                    .disabled(true)
                    .help("0381: this reason should appear on hover")
            }
        }
    }
}

struct SpikeRoot: View {
    @Environment(SpikeLog.self) private var log

    var body: some View {
        HSplitView {
            TabView {
                Tab("0381 help", systemImage: "questionmark.circle") { HelpOnDisabledItem() }
                Tab("0382 context keys", systemImage: "command") { ContextMenuKeys() }
                Tab("0383 arrows", systemImage: "arrow.up.arrow.down") { ArrowKeysFromList() }
                Tab("0384 row gaps", systemImage: "rectangle.split.1x2") { RowGaps() }
                Tab("0385 alert", systemImage: "exclamationmark.bubble") { LiveAlert() }
                Tab("0386 sections", systemImage: "sidebar.left") { CollapsingSections() }
            }
            .frame(minWidth: 520, minHeight: 420)
            List(log.lines, id: \.self) { Text($0).font(.system(.caption, design: .monospaced)) }
                .frame(minWidth: 220)
        }
    }
}

/// 0381: hover the disabled items, here and in the menu bar's Spike menu.
struct HelpOnDisabledItem: View {
    var body: some View {
        Text("Right-click here, then hover each item")
            .frame(maxWidth: .infinity, maxHeight: .infinity)
            .contentShape(Rectangle())
            .contextMenu {
                Button("Disabled With Help") {}
                    .disabled(true)
                    .help("0381: disabled reason")
                Button("Enabled With Help") {}
                    .help("0381: enabled help")
            }
    }
}

/// 0382: with no menu open, press ⌥⌘K (context menu only) and ⌥⌘J (also in
/// the menu bar). Count log lines.
struct ContextMenuKeys: View {
    @Environment(SpikeLog.self) private var log
    @State private var selection: String?

    var body: some View {
        List(["one", "two", "three"], id: \.self, selection: $selection) { Text($0) }
            .contextMenu(forSelectionType: String.self) { _ in
                Button("Context Only ⌥⌘K") { log.record("context menu: ⌥⌘K") }
                    .keyboardShortcut("k", modifiers: [.command, .option])
                Button("Shared ⌥⌘J") { log.record("context menu: Shared ⌥⌘J") }
                    .keyboardShortcut("j", modifiers: [.command, .option])
            }
    }
}

/// 0383: click a row so the list has focus, then press ⌥⌘↑ and ⌥⌘↓.
struct ArrowKeysFromList: View {
    @Environment(SpikeLog.self) private var log
    @State private var selection: String? = "c"

    var body: some View {
        List(["a", "b", "c", "d", "e"], id: \.self, selection: $selection) { Text($0) }
            .onChange(of: selection) { _, new in log.record("selection: \(new ?? "nil")") }
    }
}

/// 0384: screenshot both lists. The line must be continuous on the left one.
struct RowGaps: View {
    @State private var left: Int? = 3
    @State private var right: Int? = 3

    var body: some View {
        HStack(spacing: 0) {
            List(0..<12, id: \.self, selection: $left) { index in
                row(index)
                    .listRowInsets(EdgeInsets(top: 0, leading: 0, bottom: 0, trailing: 8))
                    .listRowSeparator(.hidden)
            }
            List(0..<12, id: \.self, selection: $right) { index in
                row(index)
            }
        }
    }

    private func row(_ index: Int) -> some View {
        HStack(spacing: 8) {
            Canvas { context, size in
                var line = Path()
                line.move(to: CGPoint(x: size.width / 2, y: 0))
                line.addLine(to: CGPoint(x: size.width / 2, y: size.height))
                context.stroke(line, with: .color(.orange), lineWidth: 2)
            }
            .frame(width: 20)
            .frame(maxHeight: .infinity)
            Text("Commit subject \(index)").lineLimit(1).padding(.vertical, 4)
        }
    }
}

/// 0385: open the alert and type; watch the button and the message.
struct LiveAlert: View {
    @State private var presented = false
    @State private var name = ""

    var body: some View {
        Button("New Branch…") { presented = true }
            .frame(maxWidth: .infinity, maxHeight: .infinity)
            .alert("New Branch", isPresented: $presented) {
                TextField("Branch name", text: $name)
                Button("Create Branch") {}
                    .disabled(name.isEmpty || name == "main")
                Button("Cancel", role: .cancel) {}
            } message: {
                Text(name == "main" ? "A branch named “main” already exists" : "Starts at a1b2c3d")
            }
    }
}

/// 0386: click each section header's disclosure control; type in the filter.
struct CollapsingSections: View {
    @State private var branchesExpanded = true
    @State private var remotesExpanded = false
    @State private var query = ""

    var body: some View {
        HSplitView {
            List {
                Section("Branches", isExpanded: $branchesExpanded) {
                    ForEach(names("issue/"), id: \.self) { Label($0, systemImage: "arrow.triangle.branch") }
                }
                Section("Remotes", isExpanded: $remotesExpanded) {
                    ForEach(names("origin/issue/"), id: \.self) { Label($0, systemImage: "network") }
                }
            }
            .listStyle(.sidebar)
            .searchable(text: $query, placement: .sidebar, prompt: "Filter")
            .frame(minWidth: 220)
            Text("branches \(branchesExpanded ? "expanded" : "collapsed"), remotes \(remotesExpanded ? "expanded" : "collapsed")")
                .frame(maxWidth: .infinity, maxHeight: .infinity)
        }
    }

    private func names(_ prefix: String) -> [String] {
        (1...40).map { "\(prefix)\(String(format: "%04d", $0))" }
            .filter { query.isEmpty || $0.localizedCaseInsensitiveContains(query) }
    }
}
```

### Who runs it

Every spike needs a person at the Mac. Hovering, pressing keys and clicking disclosure controls
cannot be synthesized by an unattended round, which has no Accessibility permission. A round may
build and launch the app; a human performs the steps and records what happened.

Window captures use `docs/windows-command.md`'s `screencapture -l` pattern and are saved under
`build/`. In each child, record: the macOS version (`sw_vers -productVersion`), the Xcode version,
what was seen, and the capture path where one is asked for.
