## Description

Split from #0556, where it was reported on #0555's large fixture: after clicking a History node, ↓
did not select the commit below. **It reproduces on the small `uitest-fixture-repo`** — #0426's
arrow stepping has never worked in the app; no spike pressed an unmodified arrow until now.

**Root cause, measured** in the VM with a probe that wrote what the map's `onKeyPress` saw into an
on-screen `Text`: after a node click the map **is** first responder (the filter field reports no
keyboard focus) and `BranchMapView`'s `.onKeyPress(keys: [.upArrow, .downArrow])` handler **is**
called, with the selection found in the layout (`found=true`, `nodes=4` small / `nodes=13` large).
But a plain arrow arrives with `press.modifiers.rawValue == 64` — **`EventModifiers.function`**
(arrow keys are function keys to AppKit) — so the handler's first guard, `press.modifiers.isEmpty`,
is false for every arrow and it returns `.ignored`.

**The fix:** step when none of ⌘ ⌥ ⌃ ⇧ is held, instead of when the set is empty — the guard's
intent (#0383: modified arrows pass through to the Commit menu's ⌥⌘↑/⌥⌘↓) is kept, and
`.function` / `.numericPad` no longer count. The test lives in a small public
`BranchMapKeys.stepsLane(_:)` so `swift test` reaches it.

**Hypotheses from #0556's report, settled:** focus — no (the map had focus). Lazy rows not realized
— no: with the fix, ↓ from `large commit 5994` (inside a fold typing had opened) selected `large
commit 5993` on the large fixture. Handler ordering — no (the handler ran).

/// #0557: which arrow presses step the map's selection through its lane.
public nonisolated enum BranchMapKeys {
    /// True for an arrow with no ⌘, ⌥, ⌃ or ⇧. An arrow key always carries
    /// `.function` (measured in the VM: a plain ↓ arrives with rawValue 64),
    /// so `modifiers.isEmpty` never held and ↓/↑ did nothing (#0426, #0557).
    public static func stepsLane(_ modifiers: EventModifiers) -> Bool {
        modifiers.isDisjoint(with: [.command, .option, .control, .shift])
    }
}

### 2. `YardKit/Tests/YardUITests/BranchMapKeysTests.swift` — new file, exactly this

```swift
// BranchMapKeysTests.swift
//
// #0557: which arrow presses step the branch map's selection. A plain arrow
// key arrives carrying `.function` (and a keypad arrow `.numericPad`); both
// must step. ⌘, ⌥, ⌃ and ⇧ pass through to the menu bar (#0383).

import SwiftUI
import Testing
import YardUI

@Test func aPlainArrowStepsTheLane() {
    #expect(BranchMapKeys.stepsLane([]))
    #expect(BranchMapKeys.stepsLane(.function))
    #expect(BranchMapKeys.stepsLane([.function, .numericPad]))
}

@Test func aModifiedArrowPassesThrough() {
    #expect(!BranchMapKeys.stepsLane([.function, .command, .option]))
    #expect(!BranchMapKeys.stepsLane([.function, .command]))
    #expect(!BranchMapKeys.stepsLane([.function, .option]))
    #expect(!BranchMapKeys.stepsLane([.function, .control]))
    #expect(!BranchMapKeys.stepsLane([.function, .shift]))
}
```

import XCTest

/// #0557: ↓ and ↑ step the History selection through its lane. A plain arrow
/// key carries `.function`, which the map's key handler read as a modifier,

final class Spike0557MapArrowsUITests: XCTestCase {
    @MainActor
    func testArrowsStepThroughTheLane() {
        let app = XCUIApplication()
        app.launchWithFixtureRepository()
        app.selectHistoryRow(subject: UITestFixture.thirdSubject)
        app.typeKey(.downArrow, modifierFlags: [])
        XCTAssertTrue(detailHeadline(app, UITestFixture.secondSubject).waitForExistence(timeout: 10),
                      "↓ did not select the commit below \"0382 third commit\"")
        app.typeKey(.upArrow, modifierFlags: [])
        XCTAssertTrue(detailHeadline(app, UITestFixture.thirdSubject).waitForExistence(timeout: 10),

        let shot = XCTAttachment(screenshot: app.windows.firstMatch.screenshot())
        shot.name = "history-arrows"
        shot.lifetime = .keepAlways
        add(shot)
    }

    /// The Detail pane's headline: the selected commit's subject on its own.
    @MainActor
    private func detailHeadline(_ app: XCUIApplication, _ subject: String) -> XCUIElement {
        app.staticTexts.matching(NSPredicate(format: "value == %@ OR label == %@", subject, subject)).firstMatch
    }
}
